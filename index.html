function calculateKPI(data){ document.getElementById("totalBooking").innerHTML = data.length; const vehicleSet = new Set(); const driverSet = new Set(); let totalCost = 0; let completed = 0; let cancelled = 0; data.forEach(row=>{ const status = String(row["Status"] || ""); if(status.includes("เสร็จ")){ completed++; } if(status.includes("ยกเลิก")){ cancelled++; } if(row["ทะเบียนรถ"]){ vehicleSet.add( row["ทะเบียนรถ"] ); } if(row["พนักงานขับรถ"]){ driverSet.add( row["พนักงานขับรถ"] ); } const cost = parseFloat( row["ค่าใช้จ่าย"] || 0 ); if(!isNaN(cost)){ totalCost += cost; } }); document.getElementById( "completedJob" ).innerHTML = completed; document.getElementById( "cancelJob" ).innerHTML = cancelled; document.getElementById( "vehicleCount" ).innerHTML = vehicleSet.size; document.getElementById( "driverCount" ).innerHTML = driverSet.size; document.getElementById( "totalCost" ).innerHTML = "฿" + totalCost.toLocaleString(); }
